4. FunctionModal (생성/편집)
   ```tsx
   export function FunctionModal({ isOpen, function: func, onClose }: Props) {
     const [step, setStep] = useState(1); // 1: 기본정보, 2: 파라미터, 3: 구현
     const [showPreview, setShowPreview] = useState(false);

     return (
       <Modal isOpen={isOpen} onClose={onClose} size="xl">
         {/* 스텝 인디케이터 */}
         <StepIndicator currentStep={step} steps={['기본 정보', '파라미터', '구현']} />

         {step === 1 && (
           <BasicInfoStep
             register={register}
             errors={errors}
             onNext={() => setStep(2)}
           />
         )}

         {step === 2 && (
           <ParameterStep
             parameters={parameters}
             onChange={setParameters}
             onNext={() => setStep(3)}
             onBack={() => setStep(1)}
           />
         )}

         {step === 3 && (
           <ImplementationStep
             implementationType={implementationType}
             implementation={implementation}
             onChange={setImplementation}
             onSubmit={handleSubmit}
             onBack={() => setStep(2)}
           />
         )}

         {/* JSON Schema 미리보기 사이드패널 */}
         {showPreview && (
           <JsonSchemaPreview
             name={watch('name')}
             description={watch('description')}
             parameters={parameters}
           />
         )}
       </Modal>
     );
   }
   ```

5. ParameterBuilder 컴포넌트
   ```tsx
   export function ParameterBuilder({ parameters, onChange }: Props) {
     const addParameter = () => {
       onChange([...parameters, {
         name: '',
         type: 'string',
         required: true,
         description: '',
       }]);
     };

     const removeParameter = (index: number) => {
       onChange(parameters.filter((_, i) => i !== index));
     };

     return (
       <div className="space-y-4">
         {parameters.map((param, index) => (
           <div key={index} className="border rounded-lg p-4">
             <div className="grid grid-cols-2 gap-4">
               <Input
                 label="파라미터명"
                 value={param.name}
                 onChange={(e) => updateParam(index, 'name', e.target.value)}
               />
               <Select
                 label="타입"
                 value={param.type}
                 onChange={(e) => updateParam(index, 'type', e.target.value)}
               >
                 <option value="string">String</option>
                 <option value="number">Number</option>
                 <option value="boolean">Boolean</option>
                 <option value="object">Object</option>
                 <option value="array">Array</option>
               </Select>
             </div>
             <Textarea
               label="설명"
               value={param.description}
               onChange={(e) => updateParam(index, 'description', e.target.value)}
             />
             <div className="flex justify-between mt-2">
               <Checkbox
                 label="필수"
                 checked={param.required}
                 onChange={(e) => updateParam(index, 'required', e.target.checked)}
               />
               <Button variant="ghost" size="sm" onClick={() => removeParameter(index)}>
                 삭제
               </Button>
             </div>
           </div>
         ))}

         <Button variant="outline" onClick={addParameter}>
           + 파라미터 추가
         </Button>
       </div>
     );
   }
   ```
