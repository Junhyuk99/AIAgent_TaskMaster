5. AgentService 구현
   ```java
   @Service
   @Transactional
   public class AgentService {

       public AgentResponse createAgent(AgentRequest req, Long userId) {
           // LLM 서버 존재 여부 확인
           LlmServer llmServer = llmServerRepository.findById(req.getLlmServerId())
               .orElseThrow(() -> new NotFoundException("LLM 서버를 찾을 수 없습니다"));

           Agent agent = Agent.builder()
               .name(req.getName())
               .description(req.getDescription())
               .systemPrompt(req.getSystemPrompt())
               .llmServer(llmServer)
               .modelName(req.getModelName())
               .temperature(req.getTemperature())
               .maxTokens(req.getMaxTokens())
               .userId(userId)
               .build();

           return toResponse(agentRepository.save(agent));
       }

       public AgentResponse duplicateAgent(Long id, String newName, Long userId) {
           Agent original = agentRepository.findByIdAndUserId(id, userId)
               .orElseThrow(() -> new NotFoundException("에이전트를 찾을 수 없습니다"));

           Agent duplicate = Agent.builder()
               .name(newName)
               .description(original.getDescription())
               .systemPrompt(original.getSystemPrompt())
               .llmServer(original.getLlmServer())
               .modelName(original.getModelName())
               .temperature(original.getTemperature())
               .maxTokens(original.getMaxTokens())
               .userId(userId)
               .build();

           // 함수 연결도 복제
           duplicate.setFunctions(new HashSet<>(original.getFunctions()));

           return toResponse(agentRepository.save(duplicate));
       }
   }
   ```
