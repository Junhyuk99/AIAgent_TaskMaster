3. API 키 생성 로직
   ```java
   @Service
   public class ApiKeyService {

       private final PasswordEncoder passwordEncoder;

       public ApiKeyCreateResponse createApiKey(ApiKeyRequest request, Long userId) {
           // API 키 생성 (prefix + random)
           String rawKey = generateApiKey();

           ApiKey apiKey = ApiKey.builder()
               .user(userRepository.findById(userId).orElseThrow())
               .name(request.getName())
               .keyPrefix(rawKey.substring(0, 12))  // "ak_" + 8자
               .keyHash(passwordEncoder.encode(rawKey))
               .expiresAt(request.getExpiresAt())
               .isActive(true)
               .usageCount(0L)
               .createdAt(LocalDateTime.now())
               .build();

           apiKeyRepository.save(apiKey);

           // 원본 키는 이때만 반환 (다시 조회 불가)
           return ApiKeyCreateResponse.builder()
               .id(apiKey.getId())
               .name(apiKey.getName())
               .key(rawKey)  // 원본 키 반환
               .keyPrefix(apiKey.getKeyPrefix())
               .expiresAt(apiKey.getExpiresAt())
               .createdAt(apiKey.getCreatedAt())
               .build();
       }

       private String generateApiKey() {
           // "ak_" + 32자 랜덤 문자열
           String random = UUID.randomUUID().toString().replace("-", "") +
                          UUID.randomUUID().toString().replace("-", "");
           return "ak_" + random.substring(0, 32);
       }

       public ApiKey validateApiKey(String rawKey) {
           // prefix로 후보 찾기
           String prefix = rawKey.substring(0, 12);
           List<ApiKey> candidates = apiKeyRepository.findByKeyPrefixAndIsActiveTrue(prefix);

           for (ApiKey candidate : candidates) {
               if (passwordEncoder.matches(rawKey, candidate.getKeyHash())) {
                   // 만료 확인
                   if (candidate.getExpiresAt() != null &&
                       candidate.getExpiresAt().isBefore(LocalDateTime.now())) {
                       throw new ApiKeyExpiredException("API 키가 만료되었습니다");
                   }

                   // 사용 통계 업데이트
                   candidate.setLastUsedAt(LocalDateTime.now());
                   candidate.setUsageCount(candidate.getUsageCount() + 1);
                   apiKeyRepository.save(candidate);

                   return candidate;
               }
           }

           throw new InvalidApiKeyException("유효하지 않은 API 키입니다");
       }
   }
   ```

4. API 키 인증 필터
   ```java
   @Component
   public class ApiKeyAuthenticationFilter extends OncePerRequestFilter {

       private final ApiKeyService apiKeyService;

       @Override
       protected void doFilterInternal(HttpServletRequest request,
                                        HttpServletResponse response,
                                        FilterChain filterChain) {

           // API 키 추출 (헤더 또는 쿼리 파라미터)
           String apiKey = extractApiKey(request);

           if (apiKey != null) {
               try {
                   ApiKey validatedKey = apiKeyService.validateApiKey(apiKey);
                   User user = validatedKey.getUser();

                   // SecurityContext에 인증 정보 설정
                   UsernamePasswordAuthenticationToken auth =
                       new UsernamePasswordAuthenticationToken(
                           user, null, user.getAuthorities());
                   auth.setDetails(new ApiKeyAuthenticationDetails(validatedKey));
                   SecurityContextHolder.getContext().setAuthentication(auth);
               } catch (Exception e) {
                   response.setStatus(HttpStatus.UNAUTHORIZED.value());
                   return;
               }
           }

           filterChain.doFilter(request, response);
       }

       private String extractApiKey(HttpServletRequest request) {
           // X-API-Key 헤더
           String headerKey = request.getHeader("X-API-Key");
           if (headerKey != null) return headerKey;

           // Authorization: Bearer 헤더
           String authHeader = request.getHeader("Authorization");
           if (authHeader != null && authHeader.startsWith("Bearer ak_")) {
               return authHeader.substring(7);
           }

           // 쿼리 파라미터 (덜 안전, 비권장)
           return request.getParameter("api_key");
       }
   }
   ```

     const handleCreate = async (data: ApiKeyRequest) => {

     return (
       <div className="p-6">
         <header>
           <h1>API 키 관리</h1>
           <Button onClick={() => setNewKeyModal(true)}>+ 새 API 키</Button>
         </header>

         <table>
           {apiKeys?.map(key => (
             <tr key={key.id}>
               <td>{key.name}</td>
               <td><code>{key.keyPrefix}...</code></td>
               <td>{key.lastUsedAt ? formatDate(key.lastUsedAt) : '사용 안함'}</td>
               <td>{key.usageCount}회</td>
               <td><DeleteButton keyId={key.id} /></td>
             </tr>
           ))}
         </table>

         {/* 생성된 키 표시 모달 (복사 가능) */}

             apiKey={generatedKey}
             onClose={() => setGeneratedKey(null)}
           />
         )}
       </div>
     );
   }
