2. OpenAPI 설정 클래스
   ```java
   @Configuration
   public class OpenApiConfig {

       @Bean
       public OpenAPI customOpenAPI() {
           return new OpenAPI()
               .info(new Info()
                   .title("AI Agent Builder API")
                   .version("1.0.0")
                   .description("AI 에이전트 빌더 플랫폼 API 문서")
                   .contact(new Contact()
                       .name("Support")
                       .email("support@example.com"))
                   .license(new License()
                       .name("MIT")
                       .url("https://opensource.org/licenses/MIT")))
               .externalDocs(new ExternalDocumentation()
                   .description("GitHub Repository")
                   .url("https://github.com/your-org/ai-agent-builder"))
               .addSecurityItem(new SecurityRequirement()
                   .addList("bearerAuth")
                   .addList("apiKey"))
               .components(new Components()
                   .addSecuritySchemes("bearerAuth", new SecurityScheme()
                       .type(SecurityScheme.Type.HTTP)
                       .scheme("bearer")
                       .bearerFormat("JWT")
                       .description("JWT 인증 토큰"))
                   .addSecuritySchemes("apiKey", new SecurityScheme()
                       .type(SecurityScheme.Type.APIKEY)
                       .in(SecurityScheme.In.HEADER)
                       .name("X-API-Key")
                       .description("API 키 인증")));
       }

       @Bean
       public GroupedOpenApi publicApi() {
           return GroupedOpenApi.builder()
               .group("public")
               .displayName("Public API")
               .pathsToMatch("/api/v1/**")
               .build();
       }

       @Bean
       public GroupedOpenApi internalApi() {
           return GroupedOpenApi.builder()
               .group("internal")
               .displayName("Internal API")
               .pathsToMatch("/api/**")
               .pathsToExclude("/api/v1/**")
               .build();
       }
   }
   ```

5. DTO Schema 어노테이션
   ```java
   @Data
   @Schema(description = "채팅 요청")
   public class ExternalChatRequest {

       @Schema(description = "사용자 메시지", example = "안녕하세요, 오늘 날씨 어때요?", required = true)
       @NotBlank
       private String message;

       @Schema(description = "대화 ID (이전 대화 이어가기)", example = "conv_abc123")
       private String conversationId;

       @Schema(description = "대화 히스토리 (클라이언트 관리 방식)")
       private List<ExternalMessage> history;

       @Schema(description = "스트리밍 응답 여부", defaultValue = "false")
       private Boolean stream = false;
   }

   @Data
   @Schema(description = "채팅 응답")
   public class ExternalChatResponse {

       @Schema(description = "응답 ID", example = "resp_xyz789")
       private String id;

       @Schema(description = "대화 ID", example = "conv_abc123")
       private String conversationId;

       @Schema(description = "AI 응답 내용", example = "안녕하세요! 오늘 서울 날씨는 맑고 기온은 22도입니다.")
       private String content;

       @Schema(description = "참조된 지식 소스")
       private List<ExternalSource> sources;

       @Schema(description = "토큰 사용량")
       private ExternalUsage usage;

       @Schema(description = "응답 생성 시간 (Unix timestamp)", example = "1703836800000")
       private Long timestamp;
   }
   ```

                   "/swagger-ui/**",
                   "/api-docs/**",
                   "/v3/api-docs/**"
               ).permitAll()

           );
           return http.build();
