   ```java
   @Entity
   public class Conversation {
       @Id @GeneratedValue
       private Long id;

       @ManyToOne
       private Agent agent;

       @ManyToOne
       private User user;

       private String title;  // 첫 메시지 기반 자동 생성

       @OneToMany(mappedBy = "conversation", cascade = CascadeType.ALL)
       @OrderBy("createdAt ASC")
       private List<Message> messages = new ArrayList<>();

       private LocalDateTime createdAt;
       private LocalDateTime updatedAt;
   }

   @Entity
   public class Message {
       @Id @GeneratedValue
       private Long id;

       @ManyToOne
       private Conversation conversation;

       @Enumerated(EnumType.STRING)
       private MessageRole role;  // USER, ASSISTANT, SYSTEM

       @Column(columnDefinition = "TEXT")
       private String content;

       @Column(columnDefinition = "JSON")
       private String metadata;  // sources, function_calls 등

       private Integer tokenCount;

       private LocalDateTime createdAt;
   }

   public enum MessageRole {
       USER, ASSISTANT, SYSTEM
   }
   ```

   ```java
   public ChatMessageResponse chat(Long agentId, ChatMessageRequest request, Long userId) {
       Agent agent = getAgentWithRelations(agentId, userId);

       // 대화 세션 관리
       Conversation conversation;
       if (request.getConversationId() != null) {
           conversation = conversationRepository.findById(request.getConversationId())
               .orElseThrow(() -> new NotFoundException("대화를 찾을 수 없습니다"));
       } else {
           conversation = createNewConversation(agent, userId);
       }

       // 사용자 메시지 저장
       Message userMessage = Message.builder()
           .conversation(conversation)
           .role(MessageRole.USER)
           .content(request.getMessage())
           .createdAt(LocalDateTime.now())
           .build();
       messageRepository.save(userMessage);

       // LLM 호출 (기존 로직)
       ChatResponse llmResponse = // ...

       // 어시스턴트 메시지 저장
       Message assistantMessage = Message.builder()
           .conversation(conversation)
           .role(MessageRole.ASSISTANT)
           .content(llmResponse.getContent())
           .metadata(toJson(Map.of(
               "sources", ragContext.isEmpty() ? List.of() : convertToSources(ragContext),
               "functionCalls", functionCalls
           )))
           .createdAt(LocalDateTime.now())
           .build();
       messageRepository.save(assistantMessage);

       // 대화 제목 업데이트 (첫 메시지일 경우)
       if (conversation.getTitle() == null) {
           conversation.setTitle(generateTitle(request.getMessage()));
       }
       conversation.setUpdatedAt(LocalDateTime.now());
       conversationRepository.save(conversation);

       return ChatMessageResponse.builder()
           .conversationId(conversation.getId())
           .content(llmResponse.getContent())
           .sources(convertToSources(ragContext))
           .build();
   }

   private String generateTitle(String firstMessage) {
       // 첫 메시지를 기반으로 제목 생성 (최대 50자)
       String title = firstMessage.length() > 50
           ? firstMessage.substring(0, 47) + "..."
           : firstMessage;
       return title;
   }
   ```

5. 대화 내보내기
   ```java
   public ResponseEntity<Resource> exportConversation(Long agentId, Long convId, String format, Long userId) {
       Conversation conversation = getConversationWithOwnerCheck(agentId, convId, userId);

       String content;
       String contentType;
       String filename;

       if ("json".equals(format)) {
           content = objectMapper.writeValueAsString(toExportFormat(conversation));
           contentType = "application/json";
           filename = "conversation_" + convId + ".json";
       } else if ("md".equals(format)) {
           content = toMarkdown(conversation);
           contentType = "text/markdown";
           filename = "conversation_" + convId + ".md";
       } else {
           throw new BadRequestException("지원하지 않는 형식: " + format);
       }

       return ResponseEntity.ok()
           .contentType(MediaType.parseMediaType(contentType))
           .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"" + filename + "\"")
           .body(new ByteArrayResource(content.getBytes(StandardCharsets.UTF_8)));
   }
   ```

   ```tsx
   export function ChatContainer({ agentId }: Props) {
     const [conversationId, setConversationId] = useState<number | null>(null);
     const { data: conversation } = useConversation(agentId, conversationId);

     // 대화 불러오기 시 메시지 복원
     useEffect(() => {
       if (conversation) {
         setMessages(conversation.messages.map(m => ({
           role: m.role.toLowerCase(),
           content: m.content,
           sources: m.sources,
         })));
       }
     }, [conversation]);

     // 메시지 전송 시 conversationId 포함
     const sendMessage = async () => {
       const response = await chatService.sendMessage(agentId, {
         message: input,
         conversationId,
         history: messages,
       });

       // 첫 메시지면 conversationId 저장
       if (!conversationId) {
         setConversationId(response.conversationId);
       }
     };

     return (
       <div className="flex h-full">
         <ConversationList
           agentId={agentId}
           currentConvId={conversationId}
           onSelect={setConversationId}
         />
         <div className="flex-1">
           <ChatMessages messages={messages} />
           <ChatInput onSend={sendMessage} />
         </div>
       </div>
     );
   }
   ```
