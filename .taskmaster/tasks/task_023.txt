1. RagService 클래스
   ```java
   @Service
   public class RagService {

       private final EmbeddingService embeddingService;
       private final VectorStore vectorStore;
       private final KnowledgeBaseService knowledgeBaseService;

       /**
        * 쿼리에 대한 관련 컨텍스트 검색
        */
       public RagContext retrieveContext(Agent agent, String query) {
           List<KnowledgeBase> knowledgeBases = agent.getKnowledgeBases();

           if (knowledgeBases.isEmpty()) {
               return RagContext.empty();
           }

           // 쿼리 임베딩
           float[] queryEmbedding = embeddingService.embed(query);

           List<RagChunk> allChunks = new ArrayList<>();

           // 각 지식 베이스에서 검색
           for (KnowledgeBase kb : knowledgeBases) {
               String collectionName = knowledgeBaseService.getCollectionName(kb);

               List<SearchResult> results = vectorStore.search(
                   collectionName,
                   queryEmbedding,
                   5,  // Top-K per knowledge base
                   0.5 // 최소 유사도
               );

               for (SearchResult result : results) {
                   allChunks.add(RagChunk.builder()
                       .content(result.getContent())
                       .score(result.getScore())
                       .knowledgeBaseName(kb.getName())
                       .documentId((Long) result.getMetadata().get("documentId"))
                       .chunkIndex((Integer) result.getMetadata().get("chunkIndex"))
                       .build());
               }
           }

           // 전체 결과 재정렬 (점수 기준)
           allChunks.sort((a, b) -> Double.compare(b.getScore(), a.getScore()));

           // 상위 N개만 선택
           List<RagChunk> topChunks = allChunks.stream()
               .limit(5)
               .collect(Collectors.toList());

           return RagContext.builder()
               .chunks(topChunks)
               .query(query)
               .build();
       }
   }
   ```

2. RagContext 및 RagChunk
   ```java
   @Data
   @Builder
   public class RagContext {
       private List<RagChunk> chunks;
       private String query;

       public static RagContext empty() {
           return RagContext.builder()
               .chunks(Collections.emptyList())
               .query("")
               .build();
       }

       public boolean isEmpty() {
           return chunks == null || chunks.isEmpty();
       }

       public String formatForPrompt() {
           if (isEmpty()) return "";

           StringBuilder sb = new StringBuilder();
           sb.append("다음은 질문과 관련된 참고 자료입니다:\n\n");

           for (int i = 0; i < chunks.size(); i++) {
               RagChunk chunk = chunks.get(i);
               sb.append(String.format("[%d] (출처: %s)\n", i + 1, chunk.getKnowledgeBaseName()));
               sb.append(chunk.getContent());
               sb.append("\n\n");
           }

           sb.append("위 참고 자료를 바탕으로 질문에 답변해주세요. ");
           sb.append("참고 자료에 없는 내용은 '해당 정보가 제공된 자료에 없습니다'라고 답변해주세요.\n\n");

           return sb.toString();
       }
   }

   @Data
   @Builder
   public class RagChunk {
       private String content;
       private Double score;
       private String knowledgeBaseName;
       private Long documentId;
       private Integer chunkIndex;
   }
   ```

3. ChatService RAG 통합

   ```java
   @Service
   public class ChatService {

       public ChatMessageResponse chat(Long agentId, ChatMessageRequest request, Long userId) {
           Agent agent = getAgentWithRelations(agentId, userId);

           // RAG 컨텍스트 검색
           RagContext ragContext = ragService.retrieveContext(agent, request.getMessage());

           // 메시지 구성
           List<Message> messages = new ArrayList<>();

           // 시스템 프롬프트 + RAG 컨텍스트
           String systemPrompt = agent.getSystemPrompt();
           if (!ragContext.isEmpty()) {
               systemPrompt = ragContext.formatForPrompt() + "\n\n" + systemPrompt;
           }
           messages.add(new Message("system", systemPrompt));

           // 대화 히스토리
           messages.addAll(request.getHistory());

           // 사용자 메시지
           messages.add(new Message("user", request.getMessage()));

           // LLM 호출
           ChatResponse llmResponse = connector.chat(buildChatRequest(agent, messages));

           // 응답에 소스 정보 포함
           return ChatMessageResponse.builder()
               .content(llmResponse.getContent())
               .sources(convertToSources(ragContext))
               .build();
       }

       private List<SourceInfo> convertToSources(RagContext context) {
           if (context.isEmpty()) return Collections.emptyList();

           return context.getChunks().stream()
               .map(chunk -> SourceInfo.builder()
                   .knowledgeBaseName(chunk.getKnowledgeBaseName())
                   .documentId(chunk.getDocumentId())
                   .relevanceScore(chunk.getScore())
                   .preview(truncate(chunk.getContent(), 200))
                   .build())
               .collect(Collectors.toList());
       }
   }
   ```

7. 컨텍스트 윈도우 관리
   ```java
   private String buildSystemPromptWithRag(Agent agent, RagContext context) {
       String basePrompt = agent.getSystemPrompt();
       String ragPrompt = context.formatForPrompt();

       // 토큰 수 추정 (대략 4자 = 1토큰)
       int estimatedTokens = (basePrompt.length() + ragPrompt.length()) / 4;
       int maxContextTokens = agent.getMaxTokens() / 2;  // 응답을 위해 절반 남김

       if (estimatedTokens > maxContextTokens) {
           // 컨텍스트 축소
           int chunksToKeep = Math.max(1, context.getChunks().size() / 2);
           context.setChunks(context.getChunks().subList(0, chunksToKeep));
           ragPrompt = context.formatForPrompt();
       }

       return ragPrompt + "\n\n" + basePrompt;
   }
   ```
