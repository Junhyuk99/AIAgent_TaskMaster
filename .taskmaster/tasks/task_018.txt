4. KnowledgeBaseRequest DTO

   ```java
   @Data
   public class KnowledgeBaseRequest {
       @NotBlank
       @Size(max = 100)
       private String name;

       @Size(max = 500)
       private String description;

       private String embeddingModel = "nomic-embed-text";

       @Min(100) @Max(2000)
       private Integer chunkSize = 500;

       @Min(0) @Max(500)
       private Integer chunkOverlap = 50;

       private ChunkingStrategy chunkingStrategy = ChunkingStrategy.FIXED_SIZE;
   }

   public enum ChunkingStrategy {
       FIXED_SIZE,      // 고정 크기
       PARAGRAPH,       // 문단별
       SEMANTIC         // 의미 기반
   }
   ```

7. KnowledgeBaseService 구현
   ```java
   @Service
   @Transactional
   public class KnowledgeBaseService {

       public void deleteKnowledgeBase(Long id, Long userId) {
           KnowledgeBase kb = getKnowledgeBaseWithOwnerCheck(id, userId);

           // 연결된 에이전트 확인
           if (!kb.getAgents().isEmpty()) {
               throw new BadRequestException(
                   "이 지식 베이스를 사용중인 에이전트가 있습니다: " +
                   kb.getAgents().stream().map(Agent::getName).collect(Collectors.joining(", ")));
           }

           // 문서 파일 삭제
           for (Document doc : kb.getDocuments()) {
               fileStorageService.delete(doc.getFilePath());
           }

           // 벡터 DB 컬렉션 삭제
           vectorStore.deleteCollection(kb.getCollectionName());

           knowledgeBaseRepository.delete(kb);
       }

       public SearchResponse search(Long id, SearchRequest request, Long userId) {
           KnowledgeBase kb = getKnowledgeBaseWithOwnerCheck(id, userId);

           List<SearchResult> results = vectorStore.search(
               kb.getCollectionName(),
               request.getQuery(),
               request.getTopK()
           );

           return SearchResponse.builder()
               .query(request.getQuery())
               .results(results)
               .build();
       }
   }
   ```

8. SearchRequest/Response
   ```java
   @Data
   public class SearchRequest {
       @NotBlank
       private String query;

       @Min(1) @Max(20)
       private Integer topK = 5;

       private Double minScore = 0.5;  // 최소 유사도
   }

   @Data
   public class SearchResponse {
       private String query;
       private List<SearchResult> results;
   }

   @Data
   public class SearchResult {
       private String content;
       private Double score;
       private String documentName;
       private Integer chunkIndex;
       private Map<String, Object> metadata;
   }
   ```
