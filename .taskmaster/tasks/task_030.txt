3. 위젯 진입점 (index.ts)
   ```typescript
   import { createRoot } from 'react-dom/client';
   import { Widget } from './Widget';
   import type { WidgetConfig } from './types';

   declare global {
     interface Window {
       AIAgentWidget: {
         init: (config: WidgetConfig) => void;
         destroy: () => void;
       };
     }
   }

   let root: ReturnType<typeof createRoot> | null = null;
   let container: HTMLElement | null = null;

   window.AIAgentWidget = {
     init(config: WidgetConfig) {
       // 컨테이너 생성
       container = document.createElement('div');
       container.id = 'ai-agent-widget-container';
       document.body.appendChild(container);

       // React 렌더링
       root = createRoot(container);
       root.render(<Widget config={config} />);
     },

     destroy() {
       if (root) {
         root.unmount();
         root = null;
       }
       if (container) {
         container.remove();
         container = null;
       }
     },
   };
   ```

4. Widget 컴포넌트
   ```tsx
   export function Widget({ config }: { config: WidgetConfig }) {
     const [isOpen, setIsOpen] = useState(false);
     const [messages, setMessages] = useState<Message[]>([]);
     const [conversationId, setConversationId] = useState<string | null>(null);

     const sendMessage = async (content: string) => {
       const userMessage = { role: 'user', content };
       setMessages(prev => [...prev, userMessage]);

       try {
         const response = await apiClient.chat({
           baseUrl: config.baseUrl,
           apiKey: config.apiKey,
           agentId: config.agentId,
           message: content,
           conversationId,
         });

         setConversationId(response.conversationId);
         setMessages(prev => [
           ...prev,
           { role: 'assistant', content: response.content },
         ]);
       } catch (error) {
         setMessages(prev => [
           ...prev,
           { role: 'error', content: '오류가 발생했습니다. 다시 시도해주세요.' },
         ]);
       }
     };

     return (
       <>
         {/* 플로팅 버튼 */}
         <ChatBubble
           isOpen={isOpen}
           onClick={() => setIsOpen(!isOpen)}
           position={config.position}
           color={config.primaryColor}
         />

         {/* 채팅 창 */}
         {isOpen && (
           <ChatWindow
             config={config}
             messages={messages}
             onSend={sendMessage}
             onClose={() => setIsOpen(false)}
           />
         )}
       </>
     );
   }
   ```

5. ChatWindow 컴포넌트
   ```tsx
   export function ChatWindow({ config, messages, onSend, onClose }: Props) {
     const [input, setInput] = useState('');
     const messagesEndRef = useRef<HTMLDivElement>(null);

     useEffect(() => {
       messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
     }, [messages]);

     const handleSend = () => {
       if (!input.trim()) return;
       onSend(input);
       setInput('');
     };

     return (
       <div
         className="ai-widget-window"
         style={{
           [config.position?.includes('left') ? 'left' : 'right']: '20px',
           [config.position?.includes('top') ? 'top' : 'bottom']: '80px',
         }}
       >
         {/* 헤더 */}
         <div className="ai-widget-header" style={{ background: config.primaryColor }}>
           <span>{config.title || 'AI 어시스턴트'}</span>
           <button onClick={onClose}>×</button>
         </div>

         {/* 메시지 목록 */}
         <div className="ai-widget-messages">
           {config.welcomeMessage && messages.length === 0 && (
             <div className="ai-widget-welcome">{config.welcomeMessage}</div>
           )}
           {messages.map((msg, idx) => (
             <div key={idx} className={`ai-widget-message ${msg.role}`}>
               {msg.content}
             </div>
           ))}
           <div ref={messagesEndRef} />
         </div>

         {/* 입력 */}
         <div className="ai-widget-input">
           <input
             value={input}
             onChange={(e) => setInput(e.target.value)}
             onKeyDown={(e) => e.key === 'Enter' && handleSend()}
             placeholder={config.placeholder || '메시지를 입력하세요...'}
           />
           <button onClick={handleSend} style={{ background: config.primaryColor }}>
             전송
           </button>
         </div>
       </div>
     );
   }
   ```

   export function EmbedCodeGenerator({ agent, apiKey }: Props) {
     const baseUrl = window.location.origin;
     const scriptUrl = `${baseUrl}/widget/ai-agent-widget.iife.js`;

     const embedCode = `<!-- AI Agent Widget -->
   <script src="${scriptUrl}"></script>
   <script>
     AIAgentWidget.init({
       apiKey: '${apiKey}',
       agentId: ${agent.id},
       baseUrl: '${baseUrl}',
       title: '${agent.name}',
       primaryColor: '#007bff',
       position: 'bottom-right',
       welcomeMessage: '안녕하세요! 무엇을 도와드릴까요?'
     });
   </script>`;

     return (
       <div>
         <h3>임베드 코드</h3>
         <p className="text-sm text-gray-500 mb-2">
           아래 코드를 웹사이트의 &lt;body&gt; 태그 끝에 추가하세요.
         </p>
         <div className="relative">
           <pre className="bg-gray-900 text-green-400 p-4 rounded text-sm overflow-x-auto">
             {embedCode}
           </pre>
           <Button
             className="absolute top-2 right-2"
             size="sm"
             onClick={() => navigator.clipboard.writeText(embedCode)}
           >
             복사
           </Button>
         </div>
       </div>
     );
   }
