5. DocumentUploader (드래그앤드롭)
   ```tsx
   export function DocumentUploader({ knowledgeBaseId }: Props) {
     const uploadDocuments = useUploadDocuments();
     const [isDragging, setIsDragging] = useState(false);

     const handleDrop = useCallback((e: React.DragEvent) => {
       e.preventDefault();
       setIsDragging(false);

       const files = Array.from(e.dataTransfer.files);
       uploadDocuments.mutate({ knowledgeBaseId, files });
     }, [knowledgeBaseId]);

     const handleFileSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
       const files = Array.from(e.target.files || []);
       uploadDocuments.mutate({ knowledgeBaseId, files });
     };

     return (
       <div
         className={`
           border-2 border-dashed rounded-lg p-8 text-center transition-colors
           ${isDragging ? 'border-blue-500 bg-blue-50' : 'border-gray-300'}
         `}
         onDragOver={(e) => { e.preventDefault(); setIsDragging(true); }}
         onDragLeave={() => setIsDragging(false)}
         onDrop={handleDrop}
       >
         <Upload className="mx-auto h-12 w-12 text-gray-400" />
         <p className="mt-2 text-sm text-gray-600">
           파일을 여기에 드래그하거나{' '}
           <label className="text-blue-500 cursor-pointer hover:underline">
             클릭하여 선택
             <input
               type="file"
               multiple
               accept=".pdf,.txt,.docx,.doc,.md,.csv,.json"
               className="hidden"
               onChange={handleFileSelect}
             />
           </label>
         </p>
         <p className="text-xs text-gray-400 mt-1">
           PDF, TXT, DOCX, MD, CSV, JSON (최대 50MB)
         </p>

         {uploadDocuments.isPending && (
           <div className="mt-4">
             <Spinner /> 업로드 중...
           </div>
         )}
       </div>
     );
   }
   ```

   ```tsx
   export function DocumentList({ knowledgeBaseId }: Props) {
     const { data: documents, isLoading } = useDocuments(knowledgeBaseId);
     const deleteDocument = useDeleteDocument();

     return (
       <div className="mt-6">
         <h3 className="font-medium mb-4">업로드된 문서 ({documents?.length || 0})</h3>

           <thead>
             <tr className="text-left border-b">
               <th className="pb-2">파일명</th>
               <th className="pb-2">크기</th>
               <th className="pb-2">청크</th>
               <th className="pb-2">상태</th>
               <th className="pb-2"></th>
             </tr>
           </thead>
           <tbody>
             {documents?.map(doc => (
               <tr key={doc.id} className="border-b">
                 <td className="py-3">{doc.filename}</td>
                 <td className="py-3 text-gray-500">{formatBytes(doc.fileSize)}</td>
                 <td className="py-3">{doc.chunkCount || '-'}</td>
                 <td className="py-3">
                   <DocumentStatusBadge status={doc.status} progress={doc.progress} />
                 </td>
                 <td className="py-3">
                   <Button
                     variant="ghost"
                     size="sm"
                     onClick={() => deleteDocument.mutate({ knowledgeBaseId, docId: doc.id })}
                   >
                     삭제
                   </Button>
                 </td>
               </tr>
             ))}
           </tbody>
         </table>
       </div>
     );
   }

   function DocumentStatusBadge({ status, progress }: Props) {
     const config = {
       PENDING: { label: '대기중', variant: 'default' },
       PROCESSING: { label: `처리중 ${progress}%`, variant: 'warning' },
       COMPLETED: { label: '완료', variant: 'success' },
       FAILED: { label: '실패', variant: 'error' },
     };

     const { label, variant } = config[status];
     return <Badge variant={variant}>{label}</Badge>;
   }
   ```

7. SearchTestPanel 컴포넌트
   ```tsx
   export function SearchTestPanel({ knowledgeBase }: Props) {
     const [query, setQuery] = useState('');
     const [results, setResults] = useState<SearchResult[]>([]);
     const searchMutation = useSearchKnowledgeBase();

     const handleSearch = () => {
       if (!query.trim()) return;
       searchMutation.mutate(
         { knowledgeBaseId: knowledgeBase.id, query },
         { onSuccess: (data) => setResults(data.results) }
       );
     };

     return (
       <div>
         <div className="flex gap-2 mb-6">
           <Input
             placeholder="검색어를 입력하세요..."
             value={query}
             onChange={(e) => setQuery(e.target.value)}
             onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
           />
           <Button onClick={handleSearch} disabled={searchMutation.isPending}>
             검색
           </Button>
         </div>

         {results.length > 0 && (
           <div className="space-y-4">
             <p className="text-sm text-gray-500">{results.length}개 결과</p>
             {results.map((result, idx) => (
               <div key={idx} className="border rounded-lg p-4">
                 <div className="flex justify-between mb-2">
                   <Badge>유사도: {(result.score * 100).toFixed(1)}%</Badge>
                 </div>
                 <p className="text-sm">{result.content}</p>
               </div>
             ))}
           </div>
         )}
       </div>
     );
   }
   ```
