2. LLM 설정 페이지 (/settings/llm)
   ```tsx
   export function LlmSettingsPage() {
     const { data: servers, isLoading } = useLlmServers();
     const [isModalOpen, setIsModalOpen] = useState(false);
     const [editingServer, setEditingServer] = useState<LlmServer | null>(null);

     return (
       <div className="p-6">
         <header className="flex justify-between items-center mb-6">
           <div>
             <h1 className="text-2xl font-bold">LLM 서버 설정</h1>
             <p className="text-gray-500">로컬 또는 원격 LLM 서버를 연결하세요</p>
           </div>
           <Button onClick={() => setIsModalOpen(true)}>
             + 서버 추가
           </Button>
         </header>

         <LlmServerList
           servers={servers}
           onEdit={setEditingServer}
           isLoading={isLoading}
         />

         <LlmServerModal
           isOpen={isModalOpen || !!editingServer}
           server={editingServer}
           onClose={() => { setIsModalOpen(false); setEditingServer(null); }}
         />
       </div>
     );
   }
   ```

3. LlmServerCard 컴포넌트
   ```tsx
   export function LlmServerCard({ server }: { server: LlmServer }) {
     const [showModels, setShowModels] = useState(false);
     const testConnection = useTestConnection();
     const deleteServer = useDeleteLlmServer();

     return (
       <div className="bg-white dark:bg-gray-800 rounded-lg shadow p-5">
         <div className="flex justify-between items-start">
           <div>
             <div className="flex items-center gap-2">
               <h3 className="font-semibold text-lg">{server.name}</h3>
               <Badge variant={server.isActive ? 'success' : 'default'}>
                 {server.isActive ? '활성' : '비활성'}
               </Badge>
             </div>
             <p className="text-sm text-gray-500 mt-1">{server.baseUrl}</p>
             <Badge variant="outline" className="mt-2">
               {server.type}
             </Badge>
           </div>

           <DropdownMenu>
             <DropdownItem onClick={() => onEdit(server)}>편집</DropdownItem>
             <DropdownItem onClick={() => testConnection.mutate(server.id)}>
               연결 테스트
             </DropdownItem>
             <DropdownItem onClick={() => deleteServer.mutate(server.id)} variant="danger">
               삭제
             </DropdownItem>
           </DropdownMenu>
         </div>

         {/* 연결 테스트 결과 */}
         <ConnectionTestResult serverId={server.id} />

         {/* 모델 목록 토글 */}
         <button onClick={() => setShowModels(!showModels)} className="mt-4">
           {showModels ? '모델 숨기기' : '사용 가능한 모델 보기'} ({server.modelCount})
         </button>

         {showModels && <ModelList serverId={server.id} />}
       </div>
     );
   }
   ```

5. ConnectionTest 컴포넌트
   ```tsx
   export function ConnectionTestResult({ serverId }: Props) {
     const { data: result, isLoading, refetch } = useConnectionTest(serverId);

     if (!result) return null;

     return (
       <div className={`mt-3 p-3 rounded ${result.success ? 'bg-green-50' : 'bg-red-50'}`}>
         <div className="flex items-center gap-2">
           {result.success ? (
             <CheckCircle className="text-green-500" />
           ) : (
             <XCircle className="text-red-500" />
           )}
           <span>{result.success ? '연결 성공' : '연결 실패'}</span>
         </div>
         {result.error && <p className="text-sm text-red-600 mt-1">{result.error}</p>}
       </div>
     );
   }
   ```
