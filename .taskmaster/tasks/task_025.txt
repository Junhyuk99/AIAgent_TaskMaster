1. 백엔드 API
   ```java
   @RestController
   @RequestMapping("/api/agents")
   public class AgentController {

       @GetMapping("/{id}/knowledge-bases")
       public List<KnowledgeBaseSummary> getAgentKnowledgeBases(@PathVariable Long id);

       @PutMapping("/{id}/knowledge-bases")
       public AgentResponse updateAgentKnowledgeBases(
           @PathVariable Long id,
           @RequestBody UpdateKnowledgeBasesRequest request);
   }

   @Data
   public class UpdateKnowledgeBasesRequest {
       @NotNull
       private List<Long> knowledgeBaseIds;
   }

   @Data
   public class KnowledgeBaseSummary {
       private Long id;
       private String name;
       private String description;
       private Integer documentCount;
       private Integer totalChunks;
   }
   ```

2. AgentService 지식 베이스 연결 로직
   ```java
   @Transactional
   public AgentResponse updateAgentKnowledgeBases(Long agentId,
                                                   List<Long> kbIds,
                                                   Long userId) {
       Agent agent = getAgentWithOwnerCheck(agentId, userId);

       // 모든 지식 베이스 ID가 사용자 소유인지 확인
       List<KnowledgeBase> knowledgeBases =
           knowledgeBaseRepository.findAllByIdInAndUserId(kbIds, userId);

       if (knowledgeBases.size() != kbIds.size()) {
           throw new BadRequestException("일부 지식 베이스를 찾을 수 없습니다");
       }

       // 기존 연결 제거 후 새로 설정
       agent.getKnowledgeBases().clear();
       agent.getKnowledgeBases().addAll(new HashSet<>(knowledgeBases));

       return toResponse(agentRepository.save(agent));
   }
   ```

3. 프론트엔드 KnowledgeTab 컴포넌트
   ```tsx
   // src/components/agents/tabs/KnowledgeTab.tsx
   export function KnowledgeTab({ agent }: Props) {
     const { data: allKnowledgeBases } = useKnowledgeBases();
     const updateKnowledgeBases = useUpdateAgentKnowledgeBases();

     const [selectedIds, setSelectedIds] = useState<Set<number>>(
       new Set(agent.knowledgeBases.map(kb => kb.id))
     );

     const handleToggle = (kbId: number) => {
       setSelectedIds(prev => {
         const next = new Set(prev);
         if (next.has(kbId)) {
           next.delete(kbId);
         } else {
           next.add(kbId);
         }
         return next;
       });
     };

     const handleSave = () => {
       updateKnowledgeBases.mutate({
         agentId: agent.id,
         knowledgeBaseIds: Array.from(selectedIds),
       });
     };

     return (
       <div className="space-y-6">
         <div className="flex justify-between items-center">
           <div>
             <p className="text-gray-500">
               에이전트가 참조할 지식 베이스를 선택하세요.
             </p>
             <p className="text-sm text-gray-400">
               선택된 지식 베이스의 문서를 기반으로 질문에 답변합니다 (RAG).
             </p>
           </div>
           <Button onClick={handleSave} disabled={updateKnowledgeBases.isPending}>
             저장
           </Button>
         </div>

         {allKnowledgeBases?.length === 0 ? (
           <EmptyState
             icon={<Database />}
             title="지식 베이스가 없습니다"
             description="먼저 지식 베이스를 생성하고 문서를 업로드하세요."
             action={
               <Button as={Link} to="/knowledge">
                 지식 베이스 만들기
               </Button>
             }
           />
         ) : (
           <div className="grid grid-cols-2 gap-4">
             {allKnowledgeBases?.map(kb => (
               <KnowledgeBaseSelectCard
                 key={kb.id}
                 knowledgeBase={kb}
                 isSelected={selectedIds.has(kb.id)}
                 onToggle={() => handleToggle(kb.id)}
               />
             ))}
           </div>
         )}
       </div>
     );
   }
   ```

6. React Query 훅
   ```tsx
   export function useUpdateAgentKnowledgeBases() {
     const queryClient = useQueryClient();

     return useMutation({
       mutationFn: ({ agentId, knowledgeBaseIds }: {
         agentId: number;
         knowledgeBaseIds: number[];
       }) => agentService.updateKnowledgeBases(agentId, knowledgeBaseIds),
       onSuccess: (_, { agentId }) => {
         queryClient.invalidateQueries({ queryKey: ['agent', agentId] });
         toast.success('지식 베이스 연결이 업데이트되었습니다');
       },
     });
   }
   ```
