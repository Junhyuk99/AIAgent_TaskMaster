3. FunctionRequest DTO
   ```java
   @Data
   public class FunctionRequest {
       @NotBlank
       @Size(max = 100)
       private String name;

       @NotBlank
       @Size(max = 500)
       private String description;

       @NotNull
       private List<ParameterDefinition> parameters;

       private String returnType;  // string, number, object, array

       @NotNull
       private ImplementationType implementationType;  // HTTP_API, CODE, TEMPLATE

       @NotNull
       private JsonNode implementation;  // 구현 방식별 설정
   }

   @Data
   public class ParameterDefinition {
       @NotBlank
       private String name;

       @NotNull
       private ParameterType type;  // string, number, boolean, object, array

       private boolean required = true;

       private String description;

       private JsonNode defaultValue;

       private JsonNode enumValues;  // 선택 가능한 값 목록
   }
   ```

5. JSON Schema 생성 유틸리티
   ```java
   @Component
   public class JsonSchemaGenerator {

       public JsonNode generateSchema(Function function) {
           ObjectNode schema = objectMapper.createObjectNode();
           schema.put("name", function.getName());
           schema.put("description", function.getDescription());

           ObjectNode parameters = schema.putObject("parameters");
           parameters.put("type", "object");

           ObjectNode properties = parameters.putObject("properties");
           ArrayNode required = parameters.putArray("required");

           for (ParameterDefinition param : function.getParameters()) {
               ObjectNode prop = properties.putObject(param.getName());
               prop.put("type", param.getType().getJsonType());
               prop.put("description", param.getDescription());

               if (param.isRequired()) {
                   required.add(param.getName());
               }
           }

           return schema;
       }
   }
   ```

6. FunctionService 구현
   ```java
   @Service
   @Transactional
   public class FunctionService {

       public FunctionTestResponse testFunction(Long id, FunctionTestRequest req, Long userId) {
           Function function = getFunctionWithOwnerCheck(id, userId);

           FunctionExecutor executor = executorFactory.getExecutor(function.getImplementationType());

           try {
               long startTime = System.currentTimeMillis();
               Object result = executor.execute(function, req.getParameters());
               long duration = System.currentTimeMillis() - startTime;

               return FunctionTestResponse.builder()
                   .success(true)
                   .result(result)
                   .executionTime(duration)
                   .build();
           } catch (Exception e) {
               return FunctionTestResponse.builder()
                   .success(false)
                   .error(e.getMessage())
                   .build();
           }
       }
   }
   ```

7. FunctionExecutorFactory
   ```java
   @Component
   public class FunctionExecutorFactory {
       private final Map<ImplementationType, FunctionExecutor> executors;

       public FunctionExecutor getExecutor(ImplementationType type) {
           return executors.get(type);
       }
   }
   ```
