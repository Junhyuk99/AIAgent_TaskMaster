2. 로컬 파일시스템 구현체
   ```java
   @Service
   @Profile("local")
   public class LocalFileStorageService implements FileStorageService {

       @Value("${storage.local.base-path:./uploads}")
       private String basePath;

       @PostConstruct
       public void init() {
           Path uploadPath = Paths.get(basePath);
           if (!Files.exists(uploadPath)) {
               Files.createDirectories(uploadPath);
           }
       }

       @Override
       public String store(MultipartFile file, String directory) {
           String filename = generateUniqueFilename(file.getOriginalFilename());
           Path targetPath = Paths.get(basePath, directory, filename);

           Files.createDirectories(targetPath.getParent());
           Files.copy(file.getInputStream(), targetPath, StandardCopyOption.REPLACE_EXISTING);

           return directory + "/" + filename;
       }

       @Override
       public Resource load(String filePath) {
           Path path = Paths.get(basePath, filePath);
           Resource resource = new UrlResource(path.toUri());
           if (!resource.exists()) {
               throw new FileNotFoundException("파일을 찾을 수 없습니다: " + filePath);
           }
           return resource;
       }

       @Override
       public void delete(String filePath) {
           Path path = Paths.get(basePath, filePath);
           Files.deleteIfExists(path);
       }

       private String generateUniqueFilename(String originalFilename) {
           String extension = getExtension(originalFilename);
           return UUID.randomUUID().toString() + "." + extension;
       }
   }
   ```

3. MinIO (S3 호환) 구현체
   ```java
   @Service
   @Profile("minio")
   public class MinioFileStorageService implements FileStorageService {

       private final MinioClient minioClient;

       @Value("${storage.minio.bucket}")
       private String bucketName;

       @Override
       public String store(MultipartFile file, String directory) {
           String filename = generateUniqueFilename(file.getOriginalFilename());
           String objectName = directory + "/" + filename;

           minioClient.putObject(
               PutObjectArgs.builder()
                   .bucket(bucketName)
                   .object(objectName)
                   .stream(file.getInputStream(), file.getSize(), -1)
                   .contentType(file.getContentType())
                   .build()
           );

           return objectName;
       }

       @Override
       public Resource load(String filePath) {
           InputStream stream = minioClient.getObject(
               GetObjectArgs.builder()
                   .bucket(bucketName)
                   .object(filePath)
                   .build()
           );
           return new InputStreamResource(stream);
       }

       @Override
       public void delete(String filePath) {
           minioClient.removeObject(
               RemoveObjectArgs.builder()
                   .bucket(bucketName)
                   .object(filePath)
                   .build()
           );
       }
   }
   ```

4. 파일 업로드 유효성 검증
   ```java
   @Component
   public class FileValidator {

       private static final long MAX_FILE_SIZE = 50 * 1024 * 1024;  // 50MB

       private static final Set<String> ALLOWED_EXTENSIONS = Set.of(
           "pdf", "txt", "docx", "doc", "md", "csv", "json"
       );

       private static final Set<String> ALLOWED_MIME_TYPES = Set.of(
           "application/pdf",
           "text/plain",
           "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
           "application/msword",
           "text/markdown",
           "text/csv",
           "application/json"
       );

       public void validate(MultipartFile file) {
           // 파일 크기 검증
           if (file.getSize() > MAX_FILE_SIZE) {
               throw new FileValidationException(
                   "파일 크기가 50MB를 초과합니다: " + file.getOriginalFilename());
           }

           // 확장자 검증
           String extension = getExtension(file.getOriginalFilename()).toLowerCase();
           if (!ALLOWED_EXTENSIONS.contains(extension)) {
               throw new FileValidationException(
                   "지원하지 않는 파일 형식입니다: " + extension);
           }

           // MIME 타입 검증 (Content-Type 스푸핑 방지)
           String contentType = file.getContentType();
           if (contentType != null && !ALLOWED_MIME_TYPES.contains(contentType)) {
               throw new FileValidationException(
                   "지원하지 않는 파일 타입입니다: " + contentType);
           }

           // 파일명 검증 (경로 조작 방지)
           String filename = file.getOriginalFilename();
           if (filename != null && (filename.contains("..") || filename.contains("/"))) {
               throw new FileValidationException("잘못된 파일명입니다");
           }
       }
   }
   ```

5. DocumentService 파일 업로드 처리
   ```java
   @Service
   public class DocumentService {

       @Transactional
       public DocumentResponse uploadDocument(Long kbId, MultipartFile file, Long userId) {
           KnowledgeBase kb = knowledgeBaseService.getKnowledgeBaseWithOwnerCheck(kbId, userId);

           // 파일 유효성 검증
           fileValidator.validate(file);

           // 파일 저장
           String directory = "knowledge-bases/" + kbId;
           String filePath = fileStorageService.store(file, directory);

           // Document 엔티티 생성
           Document document = Document.builder()
               .knowledgeBase(kb)
               .filename(file.getOriginalFilename())
               .filePath(filePath)
               .fileSize(file.getSize())
               .mimeType(file.getContentType())
               .status(DocumentStatus.PENDING)
               .build();

           document = documentRepository.save(document);

           // 비동기 처리 시작
           documentProcessingService.processAsync(document.getId());

           return toResponse(document);
       }

       @Transactional
       public List<DocumentResponse> uploadDocuments(Long kbId, List<MultipartFile> files, Long userId) {
           return files.stream()
               .map(file -> uploadDocument(kbId, file, userId))
               .collect(Collectors.toList());
       }
   }
   ```

6. 비동기 문서 처리
   ```java
   @Service
   public class DocumentProcessingService {

       @Async
       public void processAsync(Long documentId) {
           Document document = documentRepository.findById(documentId).orElseThrow();

           try {
               document.setStatus(DocumentStatus.PROCESSING);
               documentRepository.save(document);

               // 1. 파일 파싱 (다음 태스크)
               // 2. 청킹 (다음 태스크)
               // 3. 임베딩 (다음 태스크)
               // 4. 벡터 DB 저장 (다음 태스크)

               document.setStatus(DocumentStatus.COMPLETED);
           } catch (Exception e) {
               document.setStatus(DocumentStatus.FAILED);
               document.setErrorMessage(e.getMessage());
           }

           documentRepository.save(document);
       }
   }
   ```
