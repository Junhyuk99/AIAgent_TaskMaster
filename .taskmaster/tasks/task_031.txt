3. UsageService 구현
   ```java
   @Service
   public class UsageService {

       public UsageSummary getSummary(Long userId, String period) {
           LocalDateTime from = calculateFromDate(period);

           List<UsageLog> logs = usageLogRepository.findByUserIdAndCreatedAtAfter(userId, from);

           return UsageSummary.builder()
               .totalRequests(logs.size())
               .successfulRequests((int) logs.stream().filter(UsageLog::getSuccess).count())
               .failedRequests((int) logs.stream().filter(l -> !l.getSuccess()).count())
               .totalTokens(logs.stream().mapToInt(UsageLog::getTotalTokens).sum())
               .promptTokens(logs.stream().mapToInt(UsageLog::getPromptTokens).sum())
               .completionTokens(logs.stream().mapToInt(UsageLog::getCompletionTokens).sum())
               .averageLatency(logs.stream().mapToLong(UsageLog::getLatencyMs).average().orElse(0))
               .uniqueAgents((int) logs.stream().map(UsageLog::getAgentId).distinct().count())
               .period(period)
               .build();
       }

       public List<DailyUsage> getDailyUsage(Long userId, LocalDate from, LocalDate to) {
           return usageLogRepository.getDailyStats(userId, from.atStartOfDay(), to.plusDays(1).atStartOfDay());
       }

       @Transactional
       public void logUsage(UsageLogRequest request) {
           UsageLog log = UsageLog.builder()
               .userId(request.getUserId())
               .agentId(request.getAgentId())
               .apiKeyId(request.getApiKeyId())
               .endpoint(request.getEndpoint())
               .promptTokens(request.getPromptTokens())
               .completionTokens(request.getCompletionTokens())
               .totalTokens(request.getPromptTokens() + request.getCompletionTokens())
               .latencyMs(request.getLatencyMs())
               .success(request.getSuccess())
               .errorCode(request.getErrorCode())
               .createdAt(LocalDateTime.now())
               .build();

           usageLogRepository.save(log);
       }
   }
   ```

4. Repository 집계 쿼리
   ```java
   public interface UsageLogRepository extends JpaRepository<UsageLog, Long> {

       @Query("""
           SELECT new com.aiagent.dto.DailyUsage(
               CAST(u.createdAt AS LocalDate),
               COUNT(u),
               SUM(u.totalTokens),
               SUM(CASE WHEN u.success = true THEN 1 ELSE 0 END),
               AVG(u.latencyMs)
           )
           FROM UsageLog u
           WHERE u.userId = :userId
           AND u.createdAt BETWEEN :from AND :to
           GROUP BY CAST(u.createdAt AS LocalDate)
           ORDER BY CAST(u.createdAt AS LocalDate)
           """)
       List<DailyUsage> getDailyStats(Long userId, LocalDateTime from, LocalDateTime to);

       @Query("""
           SELECT new com.aiagent.dto.AgentUsage(
               u.agentId,
               COUNT(u),
               SUM(u.totalTokens),
               AVG(u.latencyMs)
           )
           FROM UsageLog u
           WHERE u.userId = :userId
           AND u.createdAt > :from
           GROUP BY u.agentId
           ORDER BY COUNT(u) DESC
           """)
       List<AgentUsage> getUsageByAgent(Long userId, LocalDateTime from);
   }
   ```

7. CSV 내보내기
   ```java
   public ResponseEntity<Resource> exportUsage(Long userId, LocalDate from, LocalDate to, String format) {
       List<UsageLog> logs = usageLogRepository.findByUserIdAndCreatedAtBetween(
           userId, from.atStartOfDay(), to.plusDays(1).atStartOfDay());

       if ("csv".equals(format)) {
           StringBuilder csv = new StringBuilder();
           csv.append("Date,Agent ID,Tokens,Latency (ms),Success\n");

           for (UsageLog log : logs) {
               csv.append(String.format("%s,%d,%d,%d,%s\n",
                   log.getCreatedAt().toLocalDate(),
                   log.getAgentId(),
                   log.getTotalTokens(),
                   log.getLatencyMs(),
                   log.getSuccess()));
           }

           return ResponseEntity.ok()
               .contentType(MediaType.parseMediaType("text/csv"))
               .header(HttpHeaders.CONTENT_DISPOSITION,
                   "attachment; filename=\"usage_" + from + "_" + to + ".csv\"")
               .body(new ByteArrayResource(csv.toString().getBytes()));
       }

       throw new BadRequestException("지원하지 않는 형식: " + format);
   }
   ```
